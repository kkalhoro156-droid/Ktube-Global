async function uploadVideo(){ if(!user){authPage();return;} const f=$("file").files[0]; const title=$("title").value.trim(); const description=$("desc").value.trim(); const selectedCategory=$("category").value; const tags=$("tags").value .split(",") .map(t=>t.trim()) .filter(Boolean) .slice(0,20); if(!f||!title){ $("uploadStatus").textContent="Video file aur title zaroori hain."; return; } if(!f.type.startsWith("video/")||f.size>50*1024*1024){ $("uploadStatus").textContent="Valid video select karein, maximum 50 MB."; return; } $("uploadStatus").textContent="Video uploading..."; try{ const path=user.id+"/"+Date.now()+"_"+f.name.replace(/[^a-zA-Z0-9._-]/g,"_"); const upload=await db.storage.from("ktube-videos") .upload(path,f,{contentType:f.type}); if(upload.error)throw upload.error; const videoUrl=db.storage.from("ktube-videos") .getPublicUrl(path).data.publicUrl; let imageUrl=null; const im=$("thumb").files[0]; if(im){ if(!im.type.startsWith("image/")) throw new Error("Thumbnail image select karein."); const imagePath=user.id+"/thumb_"+Date.now()+"_"+im.name.replace(/[^a-zA-Z0-9._-]/g,"_"); const imageUpload=await db.storage.from("ktube-videos") .upload(imagePath,im,{contentType:im.type}); if(imageUpload.error)throw imageUpload.error; imageUrl=db.storage.from("ktube-videos") .getPublicUrl(imagePath).data.publicUrl; } const result=await db.from("videos").insert({ user_id:user.id, title:title, description:description, category:selectedCategory, tags:tags, video_url:videoUrl, thumbnail_url:imageUrl }); if(result.error)throw result.error; $("uploadStatus").textContent="Video uploaded successfully!"; $("title").value=""; $("desc").value=""; $("tags").value=""; $("file").value=""; $("thumb").value=""; homePage(); }catch(e){ $("uploadStatus").textContent="Upload error: "+e.message; } }
